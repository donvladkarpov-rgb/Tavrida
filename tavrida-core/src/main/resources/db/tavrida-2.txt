CREATE TABLE USER_SESSIONS (
  SESSION_ID      UUID      NOT NULL PRIMARY KEY,
  USER_ID         INT       NOT NULL,
  EXPIRATION_TIME TIMESTAMP NOT NULL,
);  

CREATE INDEX IX_SESSIONS ON SESSIONS (SESSION_ID, USER_ID, EXPIRATION_TIME);

CREATE TABLE CODES (
  CODE_UID              VARCHAR(16)  NOT NULL PRIMARY KEY ,
  CODE_TYPE             CHAR(1)      NOT NULL CHECK (CODE_TYPE IN ('A','S')), -- A - AES, S - SHA1
  CODE_CONTENT          VARCHAR(256) NOT NULL,
  ALLOWED_USAGE         VARCHAR(2)   NOT NULL CHECK (ALLOWED_USAGE IN ('RW','R','-')), --RW - читать и писать, R - читать, - - действия запрещены
  VALID_FROM            TIMESTAMP    NOT NULL,
  VALID_TO              TIMESTAMP    NOT NULL
);

CREATE TABLE MASTER_PASSWORD (

);
